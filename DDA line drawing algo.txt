
#include <windows.h>
#include <GL/glut.h>
#include <iostream>
#include <cmath>
#include <cstdlib>

using namespace std;

void DDA(int x1, int y1, int x2, int y2)
{
    int dx = x2 - x1;
    int dy = y2 - y1;

    float m = (float)dy / dx;

    float x = x1;
    float y = y1;

    int steps;

    glBegin(GL_POINTS);

    // First condition: |dx| >= |dy|
    if (abs(dx) >= abs(dy))
    {
        steps = abs(dx);

        for (int i = 0; i <= steps; i++)
        {
            glVertex2i(round(x), round(y));

            x = x + 1;
            y = y + m;
        }
    }

    // Second condition: |dx| < |dy|
    else
    {
        steps = abs(dy);

        for (int i = 0; i <= steps; i++)
        {
            glVertex2i(round(x), round(y));

            x = x + (1.0 / m);
            y = y + 1;
        }
    }

    glEnd();
}

void display()
{
    glClear(GL_COLOR_BUFFER_BIT);

    glColor3f(1.0, 0.0, 0.0);
    glPointSize(3.0);

    DDA(50, 50, 400, 300);

    glFlush();
}

void init()
{
    glClearColor(1.0, 1.0, 1.0, 1.0);

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();

    gluOrtho2D(0, 500, 0, 500);
}

int main(int argc, char** argv)
{
    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(500, 500);
    glutInitWindowPosition(100, 100);

    glutCreateWindow("DDA Line Drawing Algorithm");

    init();

    glutDisplayFunc(display);

    glutMainLoop();

    return 0;
}
