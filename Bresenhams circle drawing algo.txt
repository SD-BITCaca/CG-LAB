
#include <windows.h>
#include <GL/glut.h>
#include <iostream>
#include <cmath>

using namespace std;

void BresenhamCircle(int xc, int yc, int R)
{
    // Step 1: Initialize x and y
    int x = 0;
    int y = R;

    // Step 2: Calculate initial decision parameter
    int p = 3 - 2 * R;

    glBegin(GL_POINTS);

    // Continue until x becomes greater than y
    while (x <= y)
    {
        // Draw 8 symmetric points of the circle
        glVertex2i(xc + x, yc + y);
        glVertex2i(xc - x, yc + y);
        glVertex2i(xc + x, yc - y);
        glVertex2i(xc - x, yc - y);

        glVertex2i(xc + y, yc + x);
        glVertex2i(xc - y, yc + x);
        glVertex2i(xc + y, yc - x);
        glVertex2i(xc - y, yc - x);

        // Case 1: If p < 0
        if (p < 0)
        {
            x = x + 1;
            y = y;

            p = p + 4 * x + 6;
        }

        // Case 2: If p >= 0
        else
        {
            x = x + 1;
            y = y - 1;

            p = p + 4 * (x - y) + 10;
        }
    }

    glEnd();
}

void display()
{
    glClear(GL_COLOR_BUFFER_BIT);

    glColor3f(1.0, 0.0, 0.0);
    glPointSize(3.0);

    // Center = (250, 250), Radius = 150
    BresenhamCircle(250, 250, 150);

    glFlush();
}

void init()
{
    glClearColor(1.0, 1.0, 1.0, 1.0);

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();

    gluOrtho2D(0, 500, 0, 500);
}

int main(int argc, char** argv)
{
    glutInit(&argc, argv);

    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(500, 500);
    glutInitWindowPosition(100, 100);

    glutCreateWindow("Bresenham Circle Drawing Algorithm");

    init();

    glutDisplayFunc(display);

    glutMainLoop();

    return 0;
}
